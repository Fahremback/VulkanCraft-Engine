cmake_minimum_required(VERSION 3.20)
project(vulkan_craft_sdk_surface LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(vulkan_craft_sdk CONFIG REQUIRED)
add_executable(sdk_surface_consumer main.cpp)
target_link_libraries(sdk_surface_consumer PRIVATE vulkan_craft_sdk)
add_library(sdk_example_plugin SHARED plugin.cpp)
target_link_libraries(sdk_example_plugin PRIVATE vulkan_craft_sdk)
add_custom_target(sdk_example_wasm_plugin SOURCES wasm-plugin.wat wasm-plugin.json)
